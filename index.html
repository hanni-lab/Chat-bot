Введіть 'exit' або 'quit' для виходу.\n") # Зберігаємо історію діалогу messages = [ {"role": "system", "content": "Ти — корисний, ввічливий та лаконічний ШІ-асистент на Python."} ] while True: try: user_input = input("Ви: ") if user_input.lower() in ["exit", "quit"]: print("До побачення!") break if not user_input.strip(): continue # Додаємо повідомлення користувача до історії messages.append({"role": "user", "content": user_input}) # Надсилаємо запит до моделі (наприклад, gpt-4o-mini або gpt-3.5-turbo) response = client.chat.completions.create( model="gpt-4o-mini", messages=messages, temperature=0.7 ) reply = response.choices[0].message.content print(f"\nАсистент: {reply}\n") # Додаємо відповідь ШІ до історії діалогу messages.append({"role": "assistant", "content": reply}) except Exception as e: print(f"Сталася помилка: {e}") if __name__ == "__main__": run_ai_assistant()
